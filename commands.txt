Terminal 2, local kernel, choose a non-staging port such as 44120:
cd /Users/miguel/arroba

cargo build --manifest-path apps/kernel/Cargo.toml --bin arroba-kernel
ARROBA_KERNEL_PORT=44120 \
ARROBA_RELAY_URL=ws://127.0.0.1:47000 \
ARROBA_RELAY_TOKEN=local-browser-terminal-relay-token \
cargo run --manifest-path apps/kernel/Cargo.toml --bin arroba-kernel

pnpm run dev:browser-terminal:local
